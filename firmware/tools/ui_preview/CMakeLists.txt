cmake_minimum_required(VERSION 3.20)
project(tab5_preview C CXX)
get_filename_component(APP "${CMAKE_CURRENT_SOURCE_DIR}/../../main" ABSOLUTE)
set(LV_BUILD_CONF_PATH ${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${APP}/../managed_components/lvgl__lvgl lvgl)
file(GLOB FONTS ${APP}/fonts/*.c)
# cJSON for the companion session parser, from ESP-IDF as in the host tests
if(DEFINED ENV{IDF_PATH})
    set(IDF $ENV{IDF_PATH})
else()
    set(IDF $ENV{HOME}/esp/esp-idf)
endif()
add_library(cjson STATIC ${IDF}/components/json/cJSON/cJSON.c)
target_include_directories(cjson PUBLIC ${IDF}/components/json/cJSON)
add_executable(preview preview.c checks.c ${FONTS} ${APP}/term/vt.c ${APP}/term/wcwidth.c ${APP}/term/term_view.c ${APP}/ui/ui_boot.c ${APP}/ui/ui_core.c ${APP}/ui/ui_tokens.c ${APP}/ui/ui_comp.c ${APP}/ui/ui_softkbd.c ${APP}/ui/ui_settings.c ${APP}/ui/ui_settings_fw.c ${APP}/ui/ui_fwupdate.c ${APP}/ota/fw_rules.c ${APP}/ui/ui_settings_net.c ${APP}/ui/ui_hosts.c ${APP}/ui/ui_host_form.c ${APP}/ui/ui_terminal.c ${APP}/term/term_keys.c ${APP}/servers/host_rules.c ${APP}/ui/ui_sessions.c ${APP}/ui/sessions_model.c ${APP}/ui/hotkeys.c ${APP}/ui/ui_vitals.c ${APP}/sys/vitals.c ${APP}/net/ts_host.c ${APP}/net/fast_rules.c ${APP}/sys/power_rules.c ${APP}/companion/cc_sessions.c ${APP}/companion/cc_cmd.c ${APP}/companion/cc_logview.c ${APP}/voice/voice_rules.c ${APP}/i18n/i18n.c ${APP}/settings/kbd_colour.c)
target_compile_definitions(preview PRIVATE UI_PREVIEW FIXTURE_DIR="${APP}/../test/host/fixtures")
target_include_directories(preview PRIVATE stubs ${APP})
target_compile_options(preview PRIVATE -Wno-deprecated-declarations)
target_link_libraries(preview lvgl cjson)
